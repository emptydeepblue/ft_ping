# ❌ INVALID IP ADDRESS
ping 999.999.999.999

# ⚠️ INVALID HOSTNAMES
ping abc..def
ping google..com
ping host@name
ping google,com
ping google:com
ping example..com
# 🌀 WHITESPACE IN HOSTNAME
ping " "
ping "    "
ping "%20%20%20"
ping -v "127.0.0.1   "
ping -v "    127.0.0.1"
ping "GOOGLE.COM     "
ping "google.com "
ping " google.com"
ping " google.com   "
# 🌐 FQDN
ping sub.domain.google.com
ping "google.com    extra"
# 🛡️ COMMAND INJECTION
ping "$(ls)"
ping "; rm -rf /"
ping "127.0.0.1; echo hello"
# 🔀 DIFFERENT BEHAVIOUR
ping -- -v
ping -- -v 127.0.0.1
ping -- -- -? 127.0.0.1
ping -- -4./ft_pong -- -v
